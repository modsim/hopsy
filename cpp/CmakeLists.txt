cmake_minimum_required(VERSION 3.25)

# Allow:
#
#   cmake -S cpp -B build
#
# as a completely standalone C++ build.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(
            hopsy-backend
            VERSION 2.0.0a0
            LANGUAGES CXX
    )
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

set(
        HOPSY_CPP_STANDALONE OFF
)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(HOPSY_CPP_STANDALONE ON)
endif()

option(
        HOPSY_BUILD_CPP_TESTS
        "Build the hopsy C++ tests"
        ${HOPSY_CPP_STANDALONE}
)

option(
        HOPSY_INSTALL_CPP
        "Install the hopsy C++ library and headers"
        ${HOPSY_CPP_STANDALONE}
)


# ---------------------------------------------------------------------------
# Library
# ---------------------------------------------------------------------------

add_library(hopsy)

add_library(
        hopsy::hopsy
        ALIAS hopsy
)

target_sources(
        hopsy
        PRIVATE
        src/MarkovChain.cpp
        src/HitAndRun.cpp
)

target_include_directories(
        hopsy
        PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

target_compile_features(
        hopsy
        PUBLIC
        cxx_std_20
)


# ---------------------------------------------------------------------------
# Dependencies
# ---------------------------------------------------------------------------

find_package(Eigen3 3.4 REQUIRED NO_MODULE)

target_link_libraries(
        hopsy
        PUBLIC
        Eigen3::Eigen
)


# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------

if(HOPSY_BUILD_CPP_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()


# ---------------------------------------------------------------------------
# Installation
# ---------------------------------------------------------------------------

if(HOPSY_INSTALL_CPP)

    install(
            TARGETS hopsy
            EXPORT hopsyTargets

            ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
            LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
            RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    )

    install(
            DIRECTORY include/hopsy
            DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
    )

    install(
            EXPORT hopsyTargets
            FILE hopsyTargets.cmake
            NAMESPACE hopsy::
            DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/hopsy
    )

endif()
